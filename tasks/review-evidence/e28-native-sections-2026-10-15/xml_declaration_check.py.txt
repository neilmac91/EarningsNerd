"""First bytes of each fetched golden primary document (fetch_docs.py.txt output): does it open with an XML declaration?

Usage: python -I xml_declaration_check.py DOCS_DIR. Reads local files only. Leading whitespace is skipped, as
edgartools' html() compares the decoded document text with startswith("<?xml").
"""
import hashlib
import pathlib
import sys

docs = sorted(pathlib.Path(sys.argv[1]).glob("*.html"))
opens_xml = 0
for doc in docs:
    raw = doc.read_bytes()
    head = raw.lstrip()[:5]
    xml = head == b"<?xml"
    opens_xml += xml
    print(f"{'xml ' if xml else 'html'}  {doc.name:45s} {len(raw):>10,} bytes  sha256 {hashlib.sha256(raw).hexdigest()[:12]}")
print(f"\n{opens_xml} of {len(docs)} documents open with an XML declaration")
