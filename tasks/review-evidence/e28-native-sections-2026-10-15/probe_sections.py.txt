CASES = [  # (cik, accession, form) from the eval golden set

for cik, acc, form in CASES:

            try:
                print("primary doc url:", getattr(f, "document", None) and getattr(f.document, "url", None))
            except Exception as e:

            try:
                obj = f.obj()
