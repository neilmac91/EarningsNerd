def sh(*a, **k):
    return subprocess.run(a, capture_output=True, text=True, **k)

sh("git", "-C", str(REPO), "worktree", "remove", "--force", str(WT))
r = sh("git", "-C", str(REPO), "worktree", "add", "--detach", str(WT), SHA)
assert r.returncode == 0, r.stderr
B = WT / "backend"
only = set(sys.argv[3:])
with OUT.open("w") as out:
    for mid, rel, old, new, tests in M:
        if only and mid not in only:
            continue
        path = B / rel
        text = path.read_text()
        n = text.count(old)
        if n != 1:
            out.write(json.dumps({"id": mid, "error": f"old occurs {n} times"}) + "\n"); out.flush(); continue
        path.write_text(text.replace(old, new))
        t0 = time.time()
        p = sh(PY, "-m", "pytest", "-q", "-p", "no:cacheprovider", "-n", "0", *tests, cwd=B)
        tail = [l for l in p.stdout.splitlines() if l.startswith(("FAILED", "ERROR")) or " passed" in l or " failed" in l][-12:]
        out.write(json.dumps({"id": mid, "killed": p.returncode != 0, "rc": p.returncode,
                              "secs": round(time.time() - t0, 1), "tail": tail}) + "\n"); out.flush()
        sh("git", "checkout", "--", ".", cwd=WT)
    # Control: the unmutated tree passes the same union of tests.
    allt = sorted({t for *_, ts in M for t in ts})
    p = sh(PY, "-m", "pytest", "-q", "-p", "no:cacheprovider", "-n", "0", *allt, cwd=B)
    out.write(json.dumps({"id": "control-unmutated", "passed": p.returncode == 0,
                          "tail": [l for l in p.stdout.splitlines() if " passed" in l or " failed" in l or l.startswith("FAILED")][-8:]}) + "\n")
sh("git", "-C", str(REPO), "worktree", "remove", "--force", str(WT))
