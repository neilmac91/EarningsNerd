"""E125: read each page of a render dump with the reader and with SpamAssassin 4.0.0's own HTML parser (offline).

Usage: SECRET_KEY=... OPENAI_API_KEY=... <backend python> score_pages.py <sa_html.pl> <dump.json> <out.json>  (from backend/)

The dump is render_prod.py's ({case: {"html", ...}}). For each page: the texts the reader reads as too like their
ground, with the ink and the ground, and the runs SpamAssassin's parser reads as low contrast, with its ink and ground.
They agree where both find the same inks on the same grounds and the same words (SpamAssassin may hand a text over in
pieces).
"""
import json
import subprocess
import sys

sys.path.insert(0, ".")
from tests.support.mail_styles import _SpamAssassinColours, spamassassin_contrast_problems  # noqa: E402

harness, dump_path, out_path = sys.argv[1:4]
dump = json.load(open(dump_path))
cases = [{"name": name, "page": case["html"]} for name, case in dump.items()]
run = subprocess.run(["perl", harness], input=json.dumps(cases).encode(), capture_output=True, check=True)
verdicts = {row["name"]: row for row in map(json.loads, filter(None, run.stdout.decode().split("\n")))}  # newlines alone: a text the parser reads may hold a line separator, where splitlines() would split
rows, disagreements = [], 0
for case in cases:
    reader = _SpamAssassinColours(case["page"])
    problems = spamassassin_contrast_problems(case["page"])
    sa = verdicts[case["name"]]
    flagged = [r for r in sa["runs"] if r["low"]]
    ours = {(ink, ground) for _text, ink, ground in reader.too_alike}
    theirs = {(r["ink"], r["ground"]) for r in flagged}
    words_ours = sorted(" ".join(text for text, _i, _g in reader.too_alike).split())
    words_theirs = sorted(" ".join(r["text"] for r in flagged).split())
    agrees = ours == theirs and words_ours == words_theirs and not reader.refused
    disagreements += not agrees
    rows.append({"name": case["name"], "reader": problems, "spamassassin_low_contrast": sa["low_contrast"],
                 "spamassassin_invalid_colour": sa["invalid_colour"], "spamassassin_flagged": flagged, "agrees": agrees})
    seen = ", ".join(f"{' '.join(r['text'].split())!r} {r['ink']} on {r['ground']}" for r in flagged) or "clear"
    print(f"{'agrees ' if agrees else 'DIFFERS'} {case['name']:28} {seen}")
json.dump(rows, open(out_path, "w"), indent=1, ensure_ascii=False)
low = sum(row["spamassassin_low_contrast"] for row in rows)
print(f"{len(rows)} pages; SpamAssassin reads {low} as low contrast; {disagreements} where the reader reads one otherwise")
sys.exit(1 if disagreements else 0)
