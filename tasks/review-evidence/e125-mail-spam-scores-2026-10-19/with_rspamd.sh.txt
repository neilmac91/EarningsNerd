#!/usr/bin/env bash
# Run one command against a bounded rspamd 4.2.2 (E110 offline measurement): start the daemon in the background,
# wait for /ping, run "$@", then stop that daemon by its PID. Nothing outlives this script.
set -u
rspamd -f -i -u root -g root > "${RSPAMD_LOG:-rspamd-bounded.out}" 2>&1 &
pid=$!
trap 'kill "$pid" 2>/dev/null; wait "$pid" 2>/dev/null' EXIT
for _ in $(seq 1 120); do
  curl -s -m 1 http://127.0.0.1:11333/ping 2>/dev/null | grep -q pong && break
  sleep 0.5
done
curl -s -m 1 http://127.0.0.1:11333/ping | grep -q pong || { echo "rspamd did not start"; exit 1; }
"$@"
