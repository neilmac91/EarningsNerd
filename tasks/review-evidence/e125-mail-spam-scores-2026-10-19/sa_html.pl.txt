# E125: run SpamAssassin 4.0's own HTML parser (Mail::SpamAssassin::HTML) over pages, offline, as
# Message::Node::rendered runs it on a mail's HTML part by default (normalize_charset 1: the part decoded to
# characters, the parser built with new(1, 1)). Reports for each page whether it reads a text as low contrast
# (font_low_contrast) or a colour as invalid (font_invalid_color), and each text run it reads, with the ink and the
# ground it reads and whether that run alone is low contrast or invalid. Input: a JSON array of {"name", "page"}
# (UTF-8); output: one JSON object a line (UTF-8).
use strict; use warnings;
use JSON::PP;
use Mail::SpamAssassin::HTML;
no warnings 'redefine';
binmode STDOUT, ':encoding(UTF-8)';
my @runs;
my $orig = \&Mail::SpamAssassin::HTML::html_font_invisible;
*Mail::SpamAssassin::HTML::html_font_invisible = sub {
  my ($self, $text) = @_;
  my @keys = qw(font_low_contrast font_invalid_color);
  my %before = map { $_ => $self->{results}{$_} } @keys;
  delete $self->{results}{$_} for @keys;
  my $r = $orig->(@_);
  my %run = map { $_ => ($self->{results}{$_} ? 1 : 0) } @keys;
  for (@keys) { $self->{results}{$_} = 1 if $before{$_} || $run{$_}; }
  my $st = $self->{text_style}[-1];
  push @runs, { ink => $st->{fgcolor}, ground => $st->{bgcolor}, text => $text, invisible => $r ? 1 : 0,
                low => $run{font_low_contrast}, invalid => $run{font_invalid_color} };
  return $r;
};
local $/; my $cases = JSON::PP->new->utf8->decode(<STDIN>);
my $json = JSON::PP->new->canonical;
for my $case (@$cases) {
  @runs = ();
  my $html = Mail::SpamAssassin::HTML->new(1, 1);
  $html->parse($case->{page});
  my $results = $html->get_results;
  print $json->encode({ name => $case->{name}, low_contrast => $results->{font_low_contrast} ? 1 : 0,
                        invalid_colour => $results->{font_invalid_color} ? 1 : 0, runs => [@runs] }), "\n";
}
