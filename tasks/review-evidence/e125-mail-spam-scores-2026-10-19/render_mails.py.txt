"""Render every mail email_service sends, with production-shaped inputs, for one revision's backend (E110 measurement).

Usage: python render_prod.py <backend_dir> <out_json>

Links are shaped as production builds them (app/routers/auth.py, invite_service.build_invite_link,
waitlist_service.build_*_link): FRONTEND_URL https://www.earningsnerd.io and a 43-character token
(secrets.token_urlsafe(32)). Names: none, a short one and a long one. The data-quality report is sent as the job sends
it: main appended its hidden plain-text copy to the rendered page (data_quality_service), the head hands its text part
to send_email. Writes {case: {"to", "subject", "html", "text"}} for score_rspamd.py.
"""
import asyncio
import html as html_lib
import json
import os
import sys

backend = sys.argv[1]
out_path = sys.argv[2]
os.environ.update({
    "SECRET_KEY": "test-secret-key-must-be-long-enough-123", "OPENAI_API_KEY": "sk-test", "SKIP_REDIS_INIT": "true",
    "RESEND_API_KEY": "", "FRONTEND_URL": "https://www.earningsnerd.io", "DATABASE_URL": "sqlite:///:memory:",
})
sys.path.insert(0, backend)

from app.services import email_service  # noqa: E402

TOKEN = "Qx7v2LmN9pRtY4wZ8kHjB3cF6dGsA1eUoIyT5rVnMbX"  # 43 characters, as secrets.token_urlsafe(32)
assert len(TOKEN) == 43
SITE = "https://www.earningsnerd.io"
NAMES = {"none": None, "short": "Jane", "long": "Alexandra Montgomery-Fitzgerald"}
captured: list[dict] = []


async def fake_send(to, subject, html, from_email=None, *, text=None, idempotency_key=None, **kwargs):
    captured.append({"to": list(to), "subject": subject, "html": html, "text": text})
    return {"id": "x"}


email_service.send_email = fake_send
FILINGS = [
    {"company_name": "Apple Inc.", "ticker": "AAPL", "filing_type": "10-K", "filing_date": "2026-10-16", "filing_id": 101},
    {"company_name": "Microsoft Corporation", "ticker": "MSFT", "filing_type": "10-Q", "filing_date": "2026-10-16", "filing_id": 102},
    {"company_name": "JPMorgan Chase & Co.", "ticker": "JPM", "filing_type": "8-K", "filing_date": "2026-10-17", "filing_id": 103},
]
EARNINGS = [
    {"ticker": "AAPL", "company_name": "Apple Inc.", "time": "amc"},
    {"ticker": "MSFT", "company_name": "Microsoft Corporation", "time": "amc"},
    {"ticker": "JPM", "company_name": "JPMorgan Chase & Co.", "time": "bmo"},
]
REPORT = {
    "ticker_mismatches": [{"ticker": "JPMX", "primary": "JPM", "cik": "0000019617"}],
    "coverage_gaps": [{"ticker": "BRK.B", "concept": "Revenues", "last_fy": 2023, "last_total_assets_fy": 2025}],
    "universe_coverage": {"tracked_companies": 512, "with_summary": 498, "pct": 97.3},
}


async def sends() -> dict:
    out = {}

    async def run(key, coroutine):
        captured.clear()
        await coroutine
        out[key] = captured[-1]

    for label, name in NAMES.items():
        await run(f"welcome[{label}]", email_service.send_waitlist_welcome_email(
            to_email="reader@example.com", name=name, position=128, referral_link=f"{SITE}/waitlist?ref=K7Q2ZP9D",
            verification_link=f"{SITE}/api/waitlist/verify/{TOKEN}"))
        await run(f"referral[{label}]", email_service.send_referral_success_email(
            to_email="reader@example.com", name=name, new_position=97, referral_link=f"{SITE}/waitlist?ref=K7Q2ZP9D"))
        await run(f"verify[{label}]", email_service.send_verification_email(
            to_email="reader@example.com", name=name, verification_link=f"{SITE}/verify-email?token={TOKEN}"))
        await run(f"invite[{label}]", email_service.send_invite_email(
            to_email="reader@example.com", name=name, magic_link=f"{SITE}/register?invite={TOKEN}"))
        await run(f"reset[{label}]", email_service.send_password_reset_email(
            to_email="reader@example.com", name=name, reset_link=f"{SITE}/reset-password?token={TOKEN}"))
        await run(f"oauth[{label}]", email_service.send_oauth_linked_email(
            to_email="reader@example.com", name=name, provider="Google"))
        await run(f"exists[{label}]", email_service.send_account_exists_email(
            to_email="reader@example.com", name=name, login_link=f"{SITE}/login", reset_link=f"{SITE}/forgot-password"))
        await run(f"alert[{label}]", email_service.send_new_filing_alert(to_email="reader@example.com", name=name, **FILINGS[0]))
        await run(f"alert-wording[{label}]", email_service.send_new_filing_alert(
            to_email="reader@example.com", name=name, **FILINGS[0], wording_changes={"items_total": 3, "has_changes": True}))
        await run(f"digest1[{label}]", email_service.send_daily_digest(to_email="reader@example.com", name=name, items=FILINGS[:1]))
        await run(f"digest3[{label}]", email_service.send_daily_digest(to_email="reader@example.com", name=name, items=FILINGS))
        await run(f"earnings1[{label}]", email_service.send_earnings_day_alert(to_email="reader@example.com", name=name, items=EARNINGS[:1]))
        await run(f"earnings3[{label}]", email_service.send_earnings_day_alert(to_email="reader@example.com", name=name, items=EARNINGS))
    page, text = email_service.render_data_quality_report(REPORT)
    if hasattr(email_service, "_wrap_text"):  # the head: the job hands the text part to send_email
        out["data-quality"] = {"to": ["ops@earningsnerd.io"], "subject": "EarningsNerd data-quality report", "html": page, "text": text}
    else:
        out["data-quality"] = {"to": ["ops@earningsnerd.io"], "subject": "EarningsNerd data-quality report",
                               "html": f'{page}<pre style="display:none">{html_lib.escape(text)}</pre>', "text": None}
    return out


result = asyncio.run(sends())
json.dump(result, open(out_path, "w"), indent=1, ensure_ascii=False)
print("wrote", len(result), "mails to", out_path)
