"""E125: score every CONTRASTS plant with SpamAssassin 4.0.0's own HTML parser as it reads a mail (sa_html.pl: the text decoded to characters, new(1, 1)), beside the reader (offline).

Usage: SECRET_KEY=... OPENAI_API_KEY=... <backend python> score_plants.py <sa_html.pl> <out.json>  (from backend/)

Each plant's page is the one the test reads (``_contrast_page``). A broken plant (one the reader finds too like its
ground) SpamAssassin must read as low contrast (font_low_contrast); a kept plant it must not; a refused plant is the
reader's refusal, and SpamAssassin's verdict on it is recorded beside it.
"""
import json
import subprocess
import sys

sys.path.insert(0, ".")
from tests.support.mail_styles import spamassassin_contrast_problems  # noqa: E402
from tests.unit.test_mail_rendered_addresses import CONTRASTS, _contrast_page  # noqa: E402

harness, out_path = sys.argv[1], sys.argv[2]
cases = [{"name": name, "page": _contrast_page(fragment), "expected": expected} for name, fragment, expected in CONTRASTS]
run = subprocess.run(["perl", harness], input=json.dumps(cases).encode(), capture_output=True, check=True)
verdicts = {row["name"]: row for row in map(json.loads, filter(None, run.stdout.decode().split("\n")))}  # newlines alone: a text the parser reads may hold a line separator, where splitlines() would split
rows, disagreements = [], 0
for case in cases:
    problems = spamassassin_contrast_problems(case["page"])
    sa = verdicts[case["name"]]
    kind = "kept" if case["expected"] is None else ("broken" if case["expected"] == "too like its ground" else "refused")
    agrees = None if kind == "refused" else (bool(sa["low_contrast"]) == (kind == "broken"))
    disagreements += agrees is False
    rows.append({"name": case["name"], "kind": kind, "reader": problems, "spamassassin": sa, "agrees": agrees})
    mark = {True: "agrees", False: "DIFFERS", None: "refused"}[agrees]
    seen = "low contrast" if sa["low_contrast"] else ("invalid colour" if sa["invalid_colour"] else "clear")
    print(f"{kind:8} {mark:8} SpamAssassin: {seen:14} {case['name']}")
json.dump(rows, open(out_path, "w"), indent=1, ensure_ascii=False)
counts = {kind: sum(row["kind"] == kind for row in rows) for kind in ("broken", "refused", "kept")}
print(f"{len(rows)} plants ({counts}); {disagreements} where SpamAssassin's parser reads a broken or kept plant otherwise")
sys.exit(1 if disagreements else 0)
